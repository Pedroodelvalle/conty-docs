export const HelpThemeToggle = () => {
  const switchTheme = (theme) => {
    document
      .querySelector(`[data-theme-preference-value="${theme}"]`)
      ?.click();
  };

  return (
    <div className="conty-theme-toggle" role="group" aria-label="Alterar tema">
      <button
        className="conty-theme-toggle-light"
        type="button"
        aria-label="Usar tema claro"
        onClick={() => switchTheme("light")}
      >
        <svg viewBox="0 0 24 24" aria-hidden="true">
          <circle cx="12" cy="12" r="4" />

        </svg>
      </button>
      <button
        className="conty-theme-toggle-dark"
        type="button"
        aria-label="Usar tema escuro"
        onClick={() => switchTheme("dark")}

          <path d="M20.5 14.1A8.5 8.5 0 0 1 9.9 3.5 8.5 8.5 0 1 0 20.5 14.1Z" />

      </button>
    </div>
  );
};
